export function usePrecoHistorico(companyId: string) {
  const buscarHistorico = async (nomeMaterial: string) => {
    const { data } = await (supabase as any)
      .from('preco_historico')
      .select('preco, fornecedor, data_referencia, fonte')
      .eq('company_id', companyId)
      .ilike('nome_material', nomeMaterial)
      .order('data_referencia', { ascending: false })
      .limit(10);
    return data || [];
  };

  const registrarPreco = async (params: {
    nome_material: string;
    unidade?: string;
    preco: number;
    fornecedor?: string;
    fonte: 'cotacao' | 'recebimento' | 'manual';
    obra_id?: string;
  }) => {
    await (supabase as any).from('preco_historico').insert({
      company_id: companyId,
      ...params,
      data_referencia: new Date().toISOString().split('T')[0],
    });
  };

  return { buscarHistorico, registrarPreco };
}

import { supabase } from '@/integrations/supabase/untyped';
export interface NotificationPayload {
  company_id: string; obra_id?: string; user_id?: string;
  tipo: string; prioridade?: 'critica'|'importante'|'informativa';
  titulo: string; mensagem?: string;
  acao_url?: string; acao_label?: string;
  metadata?: Record<string, unknown>;
}
export async function createNotification(p: NotificationPayload): Promise<void> {
  try {
    await (supabase as any).from('notifications').insert({
      ...p, prioridade: p.prioridade ?? 'informativa',
    });
  } catch (err) { console.warn('[createNotification]', err); }
}

ts// Tipos
export interface InsumoCoeficiente {
  descricao: string;
  tipo: 'mao_obra' | 'material' | 'equipamento';
  coeficiente: number; // horas por unidade de serviço
  unidade: string;
}

export interface EquipeBase {
  funcaoPrincipal: string;    // função com maior coeficiente
  qtdPrincipal: number;       // unidades por equipe (derivado da proporção)
  funcaoAuxiliar: string | null;
  qtdAuxiliar: number;        // unidades por equipe
  proporcao: string;          // ex: "2 pedreiros + 1 servente"
}

export interface AmdahlParams {
  categoria: string;
  p: number;  // fração paralelizável 0-1
  f: number;  // overhead por equipe adicional 0-1
}

export interface CalendarConfig {
  diasUteis: number[]; // 0=dom, 1=seg ... 6=sab
  horasPorDia: number;
  feriados: string[];  // formato yyyy-MM-dd
}

export interface AmdahlInput {
  quantidade: number;         // m², kg, un etc
  nEquipes: number;           // número de equipes completas
  params: AmdahlParams;
  calendar: CalendarConfig;
  insumos: InsumoCoeficiente[];
  dataInicio: string;         // yyyy-MM-dd
}

export interface AmdahlResult {
  d0DiasUteis: number;        // duração com 1 equipe
  dnDiasUteis: number;        // duração com n equipes
  dataFimSugerida: string;    // yyyy-MM-dd
  eficiencia: number;         // 0-1
  equipe: EquipeBase;
  totalFuncaoPrincipal: number;  // nEquipes × qtdPrincipal
  totalFuncaoAuxiliar: number;   // nEquipes × qtdAuxiliar
}
Função derivarEquipeBase:
Recebe array de InsumoCoeficiente filtrado por tipo = 'mao_obra'.
Ordena por coeficiente decrescente — o maior é a função principal.
Calcula a proporção: proporcaoPrincipal = coefPrincipal / coefAuxiliar.
Arredonda para o inteiro mais próximo útil (ex: 1.2/0.6 = 2 → equipe = 2 principal + 1 auxiliar).
Retorna EquipeBase.
Função calcularD0:
Trabalho total função principal = quantidade × coefPrincipal (horas).
Trabalho total função auxiliar = quantidade × coefAuxiliar (horas).
Capacidade diária por equipe: qtdPrincipal × horasPorDia para a função principal.
D0 em dias úteis = max(trabalhoPrincipal / capacidadePrincipal, trabalhoAuxiliar / capacidadeAuxiliar).
O max() é o fator limitante — a tarefa só termina quando o mais lento terminar.
Função calcularAmdahl:
Dn = D0 × [(1-P) + P/n] × [1 + f×(n-1)/n]
Eficiência = D0 / (n × Dn) — quanto do potencial de n equipes está sendo usado.
Funções de calendário:
adicionarDiasUteis(dataInicio, diasUteis, calendar) — avança pelo calendário pulando fins de semana e feriados.
contarDiasUteis(dataInicio, dataFim, calendar) — conta dias úteis entre duas datas.
